% ---- Brand colors (mirrors fullerton.css :root) ----
\usepackage{xcolor}
\definecolor{fcblue}{HTML}{0F406B}
\definecolor{fcyellow}{HTML}{FFD600}
\definecolor{fcsky}{HTML}{375E9E}
\definecolor{fcgray}{HTML}{333333}
\definecolor{fcborderlight}{HTML}{DDDDDD}

% ---- Fonts (requires xelatex/lualatex + fontspec) ----
\usepackage{fontspec}
\setmainfont{Century Gothic}[
  Path = template/fonts/,
  Extension = .ttf,
  UprightFont = *,
  BoldFont = CenturyGothicBold,
  Ligatures = TeX
]

% issue, missing file), comment the block above and uncomment

% ---- Logo above title (mirrors #title-block / .fc-sidebar-logo) ----
\usepackage{graphicx}
\graphicspath{{template/}{pdf/template/}{./}}
\usepackage{titling}
\pretitle{%
  \begin{center}
  \includegraphics[width=1.1in]{fc-oie.png}\\[1em]
  \Huge\bfseries\color{fcblue}
}
\posttitle{\end{center}}

% ---- Section / heading styles (mirrors h1/h2/h3 in fullerton.css) ----
\usepackage{titlesec}

% h1-equivalent: bold blue, rule underneath (mirrors border-bottom)
\titleformat{\section}
  {\color{fcblue}\Large\bfseries}
  {\thesection}{0.75em}{}
  [{\color{fcblue}\titlerule[1pt]}]
\titlespacing*{\section}{0pt}{1.5em}{0.75em}

% h2-equivalent: bold blue, left rule (mirrors border-inline-start)
\titleformat{\subsection}[block]
  {\normalfont\color{fcblue}\bfseries\large}
  {}{0pt}
  {\color{fcblue}\vrule width 3pt\hspace{6pt}\thesubsection\ }
\titlespacing*{\subsection}{0pt}{1.25em}{0.5em}

% h3-equivalent: italic sky blue (mirrors h3 in fullerton.css)
\titleformat{\subsubsection}
  {\normalfont\color{fcsky}\itshape\bfseries}
  {\thesubsubsection}{0.5em}{}
\titlespacing*{\subsubsection}{0pt}{1em}{0.4em}

% ---- Table styling (approximate mirror of report tables) ----
\usepackage{colortbl}
\arrayrulecolor{fcborderlight}
